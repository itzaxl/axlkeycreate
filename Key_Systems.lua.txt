-- Button maker
local function Button(text, y)
	local b = Instance.new("TextButton", main)
	b.Size = UDim2.new(0.85,0,0.12,0)
	b.Position = UDim2.new(0.5,0,y,0)
	b.AnchorPoint = Vector2.new(0.5,0.5)
	b.Text = text
	b.Font = Enum.Font.GothamBold
	b.TextSize = 17
	b.TextColor3 = Color3.new(1,1,1)
	b.BackgroundColor3 = Color3.fromRGB(40,40,40)
	Instance.new("UICorner", b).CornerRadius = UDim.new(0,16)
	return b
end

local checkBtn = Button("✅ تحقق من المفتاح", 0.58)
local linkBtn  = Button("🌐 جلب رابط المفتاح", 0.73)
local buyBtn   = Button("➕ شراء مفتاح (تيليجرام)", 0.88)

-- Click animation
local function click(btn)
	TweenService:Create(btn, TweenInfo.new(0.1), {
		Size = btn.Size - UDim2.fromOffset(6,6)
	}):Play()
	task.wait(0.1)
	TweenService:Create(btn, TweenInfo.new(0.1), {
		Size = btn.Size
	}):Play()
end

--==================== KEY CHECK (FINAL) ====================--
checkBtn.MouseButton1Click:Connect(function()
	click(checkBtn)

	if box.Text == KEY then
		status.TextColor3 = Color3.fromRGB(80,255,120)
		status.Text = "✅ مفتاح صحيح، جاري تحميل السكربت..."

		task.wait(0.8)

		TweenService:Create(main, TweenInfo.new(0.4), {
			Size = UDim2.fromScale(0,0)
		}):Play()

		TweenService:Create(blur, TweenInfo.new(0.4), {Size = 0}):Play()
		task.wait(0.5)

		gui:Destroy()
		blur:Destroy()

		-- 🔥🔥🔥 التحميل الحقيقي (مدمج داخل السيستم)
		loadstring(game:HttpGet("https://pastebin.com/raw/X1JkHu1m"))()

	else
		status.TextColor3 = Color3.fromRGB(255,80,80)
		status.Text = "❌ المفتاح غير صحيح"
	end
end)

-- Copy key link
linkBtn.MouseButton1Click:Connect(function()
	click(linkBtn)
	if setclipboard then
		setclipboard(KEY_LINK)
	end
	status.TextColor3 = Color3.fromRGB(120,180,255)
	status.Text = "📋 تم نسخ رابط المفتاح"
end)

-- Telegram
buyBtn.MouseButton1Click:Connect(function()

		setclipboard(TELEGRAM)
	end
	status.TextColor3 = Color3.fromRGB(120,180,255)
	status.Text = "📨 تواصل معنا عبر تيليجرام"
end)
